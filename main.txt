// -------------------------------------------------------------------------- //
//  Execute un script CGI et renvoie sa sortie brute (en-tetes + corps).      //
//  Version bloquante : a faire passer dans la boucle poll() plus tard.       //
// -------------------------------------------------------------------------- //
std::string executeCgi(const std::string& interpreter,
                       const std::string& scriptPath,
                       const std::string& method,
                       const std::string& query,
                       const std::string& body)
{
    // -- environnement (version minimale) --
    std::vector<std::string> envStr;
    envStr.push_back("GATEWAY_INTERFACE=CGI/1.1");
    envStr.push_back("SERVER_PROTOCOL=HTTP/1.1");
    envStr.push_back("REQUEST_METHOD=" + method);
    envStr.push_back("QUERY_STRING=" + query);
    envStr.push_back("SCRIPT_FILENAME=" + scriptPath);
    envStr.push_back("REDIRECT_STATUS=200"); // requis par php-cgi
    if (method == "POST")
    {
        std::ostringstream len;
        len << body.size();
        envStr.push_back("CONTENT_LENGTH=" + len.str());
        envStr.push_back("CONTENT_TYPE=application/x-www-form-urlencoded");
    }

    std::vector<char*> envp;
    for (size_t i = 0; i < envStr.size(); ++i)
        envp.push_back(const_cast<char*>(envStr[i].c_str()));
    envp.push_back(NULL);

    char* argv[] = {
        const_cast<char*>(interpreter.c_str()),
        const_cast<char*>(scriptPath.c_str()),
        NULL
    };

    int in[2];
    int out[2];
    if (pipe(in) < 0 || pipe(out) < 0)
        return "";

    pid_t pid = fork();
    if (pid < 0)
        return "";

    if (pid == 0)
    {
        dup2(in[0], STDIN_FILENO);
        dup2(out[1], STDOUT_FILENO);
        close(in[0]); close(in[1]);
        close(out[0]); close(out[1]);
        execve(interpreter.c_str(), argv, &envp[0]);
        std::exit(1); // on n'arrive ici que si execve a echoue
    }

    close(in[0]);
    close(out[1]);

    if (!body.empty())
        write(in[1], body.c_str(), body.size());
    close(in[1]); // EOF : le script sait que le corps est termine

    std::string output;
    char buf[4096];
    ssize_t r;  // STOP_THERE
    while ((r = read(out[0], buf, sizeof(buf))) > 0)
        output.append(buf, r);
    close(out[0]);

    waitpid(pid, NULL, 0);
    return output;
}

// -------------------------------------------------------------------------- //
//  Transforme la sortie du CGI en reponse HTTP complete.                     //
//  Gere la directive "Status:" et la redirection via "Location:".            //
// -------------------------------------------------------------------------- //
std::string buildCgiResponse(const std::string& cgiOut)
{
    // 1. Separer le bloc d'en-tetes du corps (sur la premiere ligne vide)
    std::string headerBlock, cgiBody;
    std::string::size_type sep = cgiOut.find("\r\n\r\n");
    size_t sepLen = 4;
    if (sep == std::string::npos)
    {
        sep = cgiOut.find("\n\n");
        sepLen = 2;
    }
    if (sep != std::string::npos)
    {
        headerBlock = cgiOut.substr(0, sep);
        cgiBody = cgiOut.substr(sep + sepLen);
    }
    else
        cgiBody = cgiOut; // pas d'en-tetes : tout est corps

    // 2. Analyser les en-tetes du CGI
    std::string status = "200 OK"; // valeur par defaut
    std::string forwarded;          // en-tetes a renvoyer tels quels au client
    bool hasStatus = false;
    bool hasLocation = false;

    std::istringstream hs(headerBlock);
    std::string line;
    while (std::getline(hs, line))
    {
        if (!line.empty() && line[line.size() - 1] == '\r')
            line.erase(line.size() - 1);
        if (line.empty())
            continue;

        std::string::size_type colon = line.find(':');
        if (colon == std::string::npos)
            continue; // ligne mal formee, on ignore

        std::string name = line.substr(0, colon);
        std::string value = line.substr(colon + 1);

        // trim des espaces en debut de valeur
        std::string::size_type vs = value.find_first_not_of(" \t");
        value = (vs == std::string::npos) ? "" : value.substr(vs);

        // nom en minuscules pour une comparaison insensible a la casse
        std::string lower = name;
        for (size_t i = 0; i < lower.size(); ++i)
            if (lower[i] >= 'A' && lower[i] <= 'Z')
                lower[i] = lower[i] + ('a' - 'A');

        if (lower == "status")
        {
            status = value;     // ex: "404 Not Found"
            hasStatus = true;   // -> NE PAS reinjecter dans les en-tetes
        }
        else if (lower == "content-length")
            continue;           // on recalcule la notre, on jette celle du CGI
        else
        {
            if (lower == "location")
                hasLocation = true;
            forwarded += name + ": " + value + "\r\n";
        }
    }

    // 3. Redirection a l'initiative du CGI : Location sans Status -> 302
    if (hasLocation && !hasStatus)
        status = "302 Found";

    // 4. Assembler la reponse
    std::ostringstream response;
    response << "HTTP/1.1 " << status << "\r\n";
    response << forwarded;
    response << "Content-Length: " << cgiBody.size() << "\r\n";
    response << "Connection: close\r\n";
    response << "\r\n";
    response << cgiBody;
    return response.str();
}

int main(void)
{
    int server_fd = socket(AF_INET, SOCK_STREAM, 0);
    if (server_fd < 0)
    {
        std::cerr << "socket: " << std::strerror(errno) << std::endl;
        return 1;
    }

    int opt = 1;
    setsockopt(server_fd, SOL_SOCKET, SO_REUSEADDR, &opt, sizeof(opt));

    struct sockaddr_in addr = sockaddr_in(); // mis a zero sans memset
    addr.sin_family = AF_INET;
    addr.sin_addr.s_addr = INADDR_ANY; // n'importe quelle interface
    addr.sin_port = htons(8080);       // port 8080

    if (bind(server_fd, reinterpret_cast<struct sockaddr*>(&addr), sizeof(addr)) < 0)
    {
        std::cerr << "bind: " << std::strerror(errno) << std::endl;
        close(server_fd);
        return 1;
    }
    if (listen(server_fd, 10) < 0)
    {
        std::cerr << "listen: " << std::strerror(errno) << std::endl;
        close(server_fd);
        return 1;
    }

    std::cout << "Listening on http://localhost:8080" << std::endl;

    while (true)
    {
        int client_fd = accept(server_fd, NULL, NULL);
        if (client_fd < 0)
        {
            std::cerr << "accept: " << std::strerror(errno) << std::endl;
            continue;
        }

        char buffer[4096];
        ssize_t n = recv(client_fd, buffer, sizeof(buffer) - 1, 0);
        if (n <= 0)
        {
            close(client_fd);
            continue;
        }
        buffer[n] = '\0';
        std::cout << "--- Requete recue ---\n" << buffer << std::endl;

        // ------------------------------------------------------------------ //
        //  Parsing minimal : methode, chemin, query string, corps            //
        // ------------------------------------------------------------------ //
        std::string rawReq(buffer, n);
        std::string method, target;
        std::istringstream lineStream(rawReq);
        lineStream >> method >> target;

        std::string path = target, query;
        std::string::size_type q = target.find('?');
        if (q != std::string::npos)
        {
            path = target.substr(0, q);
            query = target.substr(q + 1);
        }

        std::string body;
        std::string::size_type bsep = rawReq.find("\r\n\r\n");
        if (bsep != std::string::npos)
            body = rawReq.substr(bsep + 4);

        // ------------------------------------------------------------------ //
        //  Est-ce un CGI ? Choix de l'interpreteur selon l'extension         //
        // ------------------------------------------------------------------ //
        bool isCgi = false;
        std::string interpreter;
        std::string::size_type dot = path.rfind('.');
        if (dot != std::string::npos)
        {
            std::string ext = path.substr(dot);
            if (ext == ".py")
            {
                isCgi = true;
                interpreter = "/usr/bin/python3";
            }
            else if (ext == ".php")
            {
                isCgi = true;
                interpreter = "/usr/bin/php-cgi";
            }
        }

        // ------------------------------------------------------------------ //
        //  Routage : CGI ou reponse statique                                 //
        // ------------------------------------------------------------------ //
        if (isCgi)
        {
            // "." + path mappe naivement l'URL vers un fichier local.
            // La vraie version utilisera le "root" de ta config.
            std::string scriptPath = "." + path;
            std::string cgiOut = executeCgi(interpreter, scriptPath,
                                            method, query, body);
            std::string resp = buildCgiResponse(cgiOut);
            send(client_fd, resp.c_str(), resp.size(), 0);
        }
        else
        {
            std::string htmlBody = "<h1>Hello (world) from c++98</h1>";
            std::ostringstream response;
            response << "HTTP/1.1 200 OK\r\n"
                     << "Content-Type: text/html\r\n"
                     << "Content-Length: " << htmlBody.size() << "\r\n"
                     << "Connection: close\r\n"
                     << "\r\n"
                     << htmlBody;
            std::string raw = response.str();
            send(client_fd, raw.c_str(), raw.size(), 0);
        }

        close(client_fd);
    }

    close(server_fd);
    return 0;
}
